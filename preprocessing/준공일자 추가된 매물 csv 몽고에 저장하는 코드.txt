# -----------------------------------------------------------
# [보조 함수] 층수 데이터가 날짜로 변환된 경우 복구하는 로직
# -----------------------------------------------------------
def restore_floor_string(value):
    s = str(value).strip()
    
    # 케이스 1: "06월 15일" 형태로 변해버린 경우 -> "6/15"로 복구
    if '월' in s and '일' in s:
        # "06월 15일" -> "06/15" -> "6/15" (앞의 0 제거)
        temp = s.replace('월', '/').replace('일', '').replace(' ', '')
        try:
            parts = temp.split('/')
            return f"{int(parts[0])}/{int(parts[1])}"
        except:
            return temp

    # 케이스 2: "2024-06-15" (YYYY-MM-DD) 형태로 변해버린 경우 -> "6/15"로 복구
    # (엑셀이 연도를 맘대로 붙인 경우)
    if re.match(r'\d{4}-\d{2}-\d{2}', s):
        try:
            parts = s.split('-')
            # 월(parts[1]), 일(parts[2])만 가져와서 층수로 복원
            return f"{int(parts[1])}/{int(parts[2])}"
        except:
            return s
            
    # 그 외("중층/15", "옥탑", "3" 등)는 그대로 반환
    return s
