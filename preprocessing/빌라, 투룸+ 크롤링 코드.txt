def is_target_address(address):
    if not address or not isinstance(address, str):
        return False
    
    if "인천" in address and "중구" in address:
        for dong in INCHEON_JUNGGU_DONGS:
            if dong in address: return False 

    if "서울" in address: return True
    for gu in SEOUL_GU_LIST:
        if gu in address: return True
        
    return False

BASE32 = '0123456789bcdefghjkmnpqrstuvwxyz'

def get_all_ids_raw():
    print("🚀 [1단계] 서울 인근 Geohash 스캔 시작 (원룸 + 빌라/투룸+ 통합 수집)...")
    session = requests.Session()
    session.headers.update({"User-Agent": "Mozilla/5.0"})
    
    seed_geohashes = ['wydj', 'wydh', 'wydk', 'wydm', 'wydq', 'wyds', 'wydt', 'wydw', 'wycl', 'wydg']
    collected_ids = set()
    
    def recursive_scan(geohash, property_type):
        url = f"https://apis.zigbang.com/v2/items/{property_type}?geohash={geohash}&depositMin=0&rentMin=0&domain=zigbang&checkAny=true"
        try:
            res = session.get(url, timeout=3)
            if res.status_code == 200:
                items = res.json().get('items', [])
                for item in items:
                    collected_ids.add(item['itemId'])
                
                if len(items) >= 50 and len(geohash) < 6: 
                    for char in BASE32:
                        recursive_scan(geohash + char, property_type)
        except: pass

    for seed in seed_geohashes:
        before_cnt = len(collected_ids)
        recursive_scan(seed, 'oneroom')
        recursive_scan(seed, 'villa')
        diff = len(collected_ids) - before_cnt
        print(f"   👉 구역 '{seed}' 스캔 완료 (+{diff}개 확보)")

    result_ids = list(collected_ids)
    print(f"📦 [1단계 완료] 총 {len(result_ids)}개의 매물 ID 확보")
    return result_ids

# 3. 셀레니움 설정
# -----------------------------------------------------------
def get_chrome_driver():
    options = Options()
    options.add_argument("--headless")
    options.add_argument("--no-sandbox")
    options.add_argument("--disable-dev-shm-usage")
    options.add_argument("user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/121.0.0.0 Safari/537.36")
    options.add_argument("--disable-blink-features=AutomationControlled")
    service = Service(ChromeDriverManager().install())
    driver = webdriver.Chrome(service=service, options=options)
    
    # 🔥 [중요 최적화] 무한 로딩 방지: 15초 이상 페이지가 안 뜨면 강제 중단
    driver.set_page_load_timeout(15) 
    return driver

# 4. 상세 파싱
# -----------------------------------------------------------
ROOM_MAP = {"원룸": "1개", "투룸": "2개", "쓰리룸": "3개", "포룸": "4개"}

def parse_detail_data(item_id, json_data, room_counts_val):
    item = json_data.get('item')
    if not item or not isinstance(item, dict): return None

    jibun = item.get('jibunAddress') or item.get('address')
    
    if not is_target_address(jibun):
        return None  

    item_url = f"https://www.zigbang.com/home/oneroom/items/{item_id}"
    price_info = item.get('price') or {}
    deposit = price_info.get('deposit', 0)
    rent = price_info.get('rent', 0)
    sales_type = item.get('salesType', '')
    
    if sales_type == '전세':
        price, deposit = deposit, None
    else:
        price = rent

    year_built = ""
    date_pattern = r'(\d{4})(?:\s*[-./,년]\s*(\d{1,2})(?:\s*[-./,월]\s*(\d{1,2}))?)?'
    date_keys = ['approve_date', 'approveDate', 'useApproveDate', 'buildDate', 'approvalDate']
    for key in date_keys:
        val = item.get(key)
        if val is not None:
            match = re.search(date_pattern, str(val).strip())
            if match:
                y, m, d = match.groups()
                if y and m and d: year_built = f"{y}-{int(m):02d}-{int(d):02d}"
                elif y and m: year_built = f"{y}-{int(m):02d}"
                elif y: year_built = f"{y}"
                break
                
    if not year_built:
        item_raw_string = json.dumps(item, ensure_ascii=False)
        match_a = re.search(r'(\d{4})(?:\s*[-./,년]\s*(\d{1,2})(?:\s*[-./,월]\s*(\d{1,2}))?)?\s*[일]?\s*준공', item_raw_string)
        if match_a:
            y, m, d = match_a.groups()
            if y and m and d: year_built = f"{y}-{int(m):02d}-{int(d):02d}"
            elif y and m: year_built = f"{y}-{int(m):02d}"
            elif y: year_built = f"{y}"

    options_data = item.get('options')
    options_str = ",".join([str(o) for o in options_data]) if isinstance(options_data, list) else None

    images_list = item.get('images', [])
    image_links = "|".join([str(img.get('url', '')) if isinstance(img, dict) else str(img) for img in images_list if img])

    raw_size = item.get('전용면적', {}).get('m2') or item.get('전용면적') or item.get('area', {}).get('m2') or item.get('area', {}).get('전용면적M2')
    size_val = 0.0
    if raw_size:
        try: size_val = float(re.sub(r'[^0-9.]', '', str(raw_size)))
        except: pass

    floor_raw = item.get('floor')
    all_floors_raw = item.get('floor_all')
    fl_curr = str(floor_raw.get('floor', '')).replace('층', '') if isinstance(floor_raw, dict) else str(floor_raw).replace('층', '')
    fl_all = str(floor_raw.get('allFloors', '')).replace('층', '') if isinstance(floor_raw, dict) else str(all_floors_raw).replace('층', '')
    floor_str = f"{fl_curr}/{fl_all}" if fl_all else fl_curr
    
    has_parking = "주차 불가능"
    parking_keys = ['parking', 'parkingYn', 'parking_yn', 'isParking']
    for key in parking_keys:
        val = item.get(key)
        if val is not None:
            val_str = str(val).replace(" ", "").lower()
            if "불가능" in val_str or val_str in ['0', '0.0', '0대', 'false', '없음', '-', 'none', '']: has_parking = "주차 불가능"
            else: has_parking = "주차 가능"
            break 

    if has_parking == "주차 불가능" and options_str and "주차" in options_str: has_parking = "주차 가능"
    if has_parking == "주차 불가능" and ("주차가능" in json.dumps(item, ensure_ascii=False).replace(" ", "")): has_parking = "주차 가능"

    loc = item.get('random_location') or item.get('location') or {}

    return {
        'id': item_id, 'url': item_url, 'address': jibun, 'rent_type': sales_type,
        'deposit': deposit, 'price': price, 'year_built': year_built, 'size_m2': size_val,
        'images_count': len(image_links.split('|')) if image_links else 0, 'images': image_links, 
        'floor': floor_str, 'hasParking': has_parking, 'options': options_str,
        'lat': loc.get('lat'), 'lng': loc.get('lng'), 'room_counts': room_counts_val
    }

# -----------------------------------------------------------
# 5. 메인 실행 (안정성 강화)
# -----------------------------------------------------------
def main():
    target_ids = get_all_ids_raw()
    if not target_ids:
        print("❌ 수집된 매물이 없습니다. (네트워크/차단 확인 필요)")
        return

    print(f"\n🚀 [2단계] 상세 수집 시작 (대상: {len(target_ids)}개) - 50개마다 자동 저장됩니다.")
    driver = get_chrome_driver()
    results = []
    
    # 🔥 실시간 저장을 위한 파일명 선언
    csv_filename = "seoul_zigbang_complete_with_villa.csv"
    
    # 만약 기존에 돌리다 만 파일이 있다면 이어서 쓸지 덮어쓸지 초기화
    if os.path.exists(csv_filename):
        os.remove(csv_filename) # 새로 시작할 때마다 초기화 (원치 않으면 주석 처리)

    try:
        driver.get("https://www.zigbang.com")
        time.sleep(2)
        
        for i, iid in enumerate(target_ids):
            try:
                # 1. API 기본 JSON 데이터 수집
                driver.get(f"https://apis.zigbang.com/v3/items/{iid}")
                body = driver.find_element(By.TAG_NAME, "body").text
                if not body.strip().startswith("{"):
                     try: body = driver.find_element(By.TAG_NAME, "pre").text
                     except: pass
                data = json.loads(body)

                # 2. 직방 사용자 화면으로 접속하여 '방 개수' 크롤링
                driver.get(f"https://www.zigbang.com/home/villa/items/{iid}")
                time.sleep(0.5) 
                
                room_counts_val = "1개" # 기본값
                page_source = driver.page_source
                
                # 🔥 [수정 핵심 1] 정규표현식으로 메뉴바(GNB)의 텍스트를 무시하고 실제 방 정보만 타겟팅
                # 예: "투룸 (욕실 1개)" 형태를 정확히 잡아냄
                match = re.search(r'(원룸|투룸|쓰리룸|포룸)\s*\([^)]*욕실', page_source)
                
                if match:
                    room_counts_val = ROOM_MAP[match.group(1)]
                else:
                    # 화면에서 못 찾은 경우 JSON 데이터의 'title' 속성에서 2차 검증
                    item_title = data.get('item', {}).get('title', '')
                    for keyword, mapped_val in ROOM_MAP.items():
                        if keyword in item_title:
                            room_counts_val = mapped_val
                            break
                
                # 3. 데이터 파싱
                parsed = parse_detail_data(iid, data, room_counts_val)
                
                # 4. 🔥 [수정 핵심 2] 50개 단위로 실시간 저장 (도중에 멈춰도 데이터 보존)
                if parsed:
                    results.append(parsed)
                    
                    if len(results) >= 50:
                        df = pd.DataFrame(results)
                        cols = ['id', 'rent_type', 'address', 'deposit', 'price', 'year_built', 'size_m2', 'floor', 'room_counts', 'options', 'hasParking', 'images', 'url', 'lat', 'lng']
                        final_cols = [c for c in cols if c in df.columns]
                        df = df[final_cols]
                        
                        # 파일이 없으면 헤더 포함 저장, 있으면 헤더 빼고 이어붙이기(append)
                        if not os.path.exists(csv_filename):
                            df.to_csv(csv_filename, index=False, encoding='utf-8-sig', mode='w')
                        else:
                            df.to_csv(csv_filename, index=False, encoding='utf-8-sig', mode='a', header=False)
                        
                        print(f"   ✅ {i+1}/{len(target_ids)} 탐색 중... (50개 자동 저장 완료) | 최근: {parsed['address']} [방 {room_counts_val}]")
                        results = [] # 저장 후 메모리 비우기
                
            except Exception as e:
                # 타임아웃 오류 발생 시 그냥 다음 매물로 넘어감 (멈춤 방지)
                pass

            # 🔥 [수정 핵심 3] 500개마다 브라우저 강제 재시작 (메모리 뻗음 완벽 방지)
            if (i + 1) % 500 == 0:
                print("   🔄 [시스템] 메모리 최적화를 위해 브라우저를 재시작합니다...")
                driver.quit()
                driver = get_chrome_driver()

        # 루프가 완전히 끝난 후 남은 자투리 데이터 저장
        if results:
            df = pd.DataFrame(results)
            cols = ['id', 'rent_type', 'address', 'deposit', 'price', 'year_built', 'size_m2', 'floor', 'room_counts', 'options', 'hasParking', 'images', 'url', 'lat', 'lng']
            final_cols = [c for c in cols if c in df.columns]
            df = df[final_cols]
            
            if not os.path.exists(csv_filename):
                df.to_csv(csv_filename, index=False, encoding='utf-8-sig', mode='w')
            else:
                df.to_csv(csv_filename, index=False, encoding='utf-8-sig', mode='a', header=False)
                
        print(f"\n✨ [최종 완료] 모든 작업이 끝났습니다. '{csv_filename}'을 확인하세요!")

    finally:
        driver.quit()

if __name__ == "__main__":
    main()
