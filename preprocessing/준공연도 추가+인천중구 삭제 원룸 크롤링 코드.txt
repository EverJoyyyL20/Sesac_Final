def is_target_address(address):
    """
    주소를 분석하여 수집 대상인지 확인
    1. 인천 중구(특정 동)이면 False (삭제)
    2. 서울이면 True (수집)
    """
    if not address or not isinstance(address, str):
        return False
    
    # ⛔ [통합] 인천 중구 필터링 (삭제 로직)
    if "인천" in address and "중구" in address:
        for dong in INCHEON_JUNGGU_DONGS:
            if dong in address:
                return False # 인천 중구 특정 동이면 수집 안 함

    # ✅ 서울 필터링 (수집 로직)
    if "서울" in address: return True
    for gu in SEOUL_GU_LIST:
        if gu in address: return True
        
    return False

# 2. Step 1: 무조건 수집 (주소 체크 제거)
# -----------------------------------------------------------
BASE32 = '0123456789bcdefghjkmnpqrstuvwxyz'

def get_all_ids_raw():
    print("🚀 [1단계] 서울 인근 Geohash 스캔 시작 (주소 검증은 2단계에서)...")
    
    session = requests.Session()
    session.headers.update({"User-Agent": "Mozilla/5.0"})
    
    # 서울 커버리지 Geohash
    seed_geohashes = ['wydj', 'wydh', 'wydk', 'wydm', 'wydq', 'wyds', 'wydt', 'wydw', 'wycl', 'wydg']
    collected_ids = set()
    
    def recursive_scan(geohash):
        url = f"https://apis.zigbang.com/v2/items/oneroom?geohash={geohash}&depositMin=0&rentMin=0&domain=zigbang&checkAny=true"
        try:
            res = session.get(url, timeout=3)
            if res.status_code == 200:
                items = res.json().get('items', [])
                
                # 주소 확인 없이 ID만 있으면 무조건 담음
                for item in items:
                    collected_ids.add(item['itemId'])
                
                # 매물이 많으면 더 깊게 탐색
                if len(items) >= 50 and len(geohash) < 6: 
                    for char in BASE32:
                        recursive_scan(geohash + char)
        except: pass

    for seed in seed_geohashes:
        before_cnt = len(collected_ids)
        recursive_scan(seed)
        diff = len(collected_ids) - before_cnt
        print(f"   👉 구역 '{seed}' 스캔 완료 (+{diff}개 확보)")

    result_ids = list(collected_ids)
    print(f"📦 [1단계 완료] 총 {len(result_ids)}개의 매물 ID 확보 (서울+인근 경기 포함)")
    return result_ids

# 3. 셀레니움 설정

def get_chrome_driver():
    options = Options()
    options.add_argument("--headless")
    options.add_argument("--no-sandbox")
    options.add_argument("--disable-dev-shm-usage")
    options.add_argument("user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/121.0.0.0 Safari/537.36")
    options.add_argument("--disable-blink-features=AutomationControlled")
    service = Service(ChromeDriverManager().install())
    return webdriver.Chrome(service=service, options=options)

# 4. 상세 파싱 (여기서 필터링 및 준공일자 추출 수행)
# -----------------------------------------------------------
def parse_detail_data(item_id, json_data):
    item = json_data.get('item')
    if not item or not isinstance(item, dict): return None

    # 주소
    jibun = item.get('jibunAddress') or item.get('address')
    
    # ★ [중요] 여기서 서울인지 확인하고, 인천 중구면 버림 ★
    if not is_target_address(jibun):
        return None  # None을 리턴하면 수집 리스트에 추가되지 않음

    # 링크
    item_url = f"https://www.zigbang.com/home/oneroom/items/{item_id}"

    # 가격
    price_info = item.get('price') or {}
    deposit = price_info.get('deposit', 0)
    rent = price_info.get('rent', 0)
    sales_type = item.get('salesType', '')
    if sales_type == '전세':
        price = deposit
        deposit = None
    else:
        price = rent

    # -------------------------------------------------------------
    # 🚧 [수정됨] 준공일자(year_built) 추출 로직 완벽 수정
    # (/, , 구분자 포함 및 연도만, 연월만 있는 케이스 모두 대응)
    # -------------------------------------------------------------
    year_built = ""
    
    # 정규식 패턴: 4자리 연도, (선택)구분자+월, (선택)구분자+일
    # 구분자는 -, ., /, ,, 년, 월, 공백 모두 허용
    date_pattern = r'(\d{4})(?:\s*[-./,년]\s*(\d{1,2})(?:\s*[-./,월]\s*(\d{1,2}))?)?'

    # 1차 시도: API에 존재할 수 있는 다양한 승인일자 관련 키를 모두 확인
    date_keys = ['approve_date', 'approveDate', 'useApproveDate', 'buildDate', 'approvalDate']
    for key in date_keys:
        val = item.get(key)
        if val is not None:
            val_str = str(val).strip()
            match = re.search(date_pattern, val_str)
            if match:
                y, m, d = match.groups()
                # 추출된 정보(연/월/일)가 있는 만큼만 조립해서 변환
                if y and m and d:
                    year_built = f"{y}-{int(m):02d}-{int(d):02d}"
                elif y and m:
                    year_built = f"{y}-{int(m):02d}"
                elif y:
                    year_built = f"{y}"
                break
                
    # 2차 시도 (최후 수단): API 구조가 특이해서 값을 못 찾았다면, 전체 JSON 텍스트에서 직접 탐색
    if not year_built:
        item_raw_string = json.dumps(item, ensure_ascii=False)
        
        # 패턴 A: "2010.10.28 준공", "2017/4/18 준공", "2020년 5월 준공", "2015 준공" 등 화면 렌더링 텍스트 탐색
        pattern_a = r'(\d{4})(?:\s*[-./,년]\s*(\d{1,2})(?:\s*[-./,월]\s*(\d{1,2}))?)?\s*[일]?\s*준공'
        match_a = re.search(pattern_a, item_raw_string)
        if match_a:
            y, m, d = match_a.groups()
            if y and m and d: year_built = f"{y}-{int(m):02d}-{int(d):02d}"
            elif y and m: year_built = f"{y}-{int(m):02d}"
            elif y: year_built = f"{y}"
        else:
            # 패턴 B: 깊숙이 숨겨진 키값 탐색 ("approveDate":"2010/10" 등)
            pattern_b = r'"(?:approve_date|approveDate|useApproveDate|buildDate)"\s*:\s*"(\d{4})(?:\s*[-./,년]\s*(\d{1,2})(?:\s*[-./,월]\s*(\d{1,2}))?)?'
            match_b = re.search(pattern_b, item_raw_string, re.IGNORECASE)
            if match_b:
                y, m, d = match_b.groups()
                if y and m and d: year_built = f"{y}-{int(m):02d}-{int(d):02d}"
                elif y and m: year_built = f"{y}-{int(m):02d}"
                elif y: year_built = f"{y}"
    # -------------------------------------------------------------

    # 옵션
    options_data = item.get('options')
    options_str = None
    if options_data and isinstance(options_data, list):
        options_str = ",".join([str(o) for o in options_data])

    # 이미지
    images_list = item.get('images', [])
    image_links = ""
    if isinstance(images_list, list):
        urls = []
        for img in images_list:
            if isinstance(img, dict): urls.append(str(img.get('url', '')))
            elif isinstance(img, str): urls.append(img)
        image_links = "|".join([u for u in urls if u])

    # 면적
    raw_size = None
    val_1 = item.get('전용면적')
    if isinstance(val_1, dict): raw_size = val_1.get('m2')
    elif val_1: raw_size = val_1
        
    if not raw_size:
        val_area = item.get('area')
        if isinstance(val_area, dict):
            raw_size = val_area.get('m2')
            if not raw_size: raw_size = val_area.get('전용면적M2')
            if not raw_size: raw_size = val_area.get('계약면적M2')
        elif val_area:
            raw_size = val_area

    if not raw_size:
        val_2 = item.get('계약면적')
        if isinstance(val_2, dict): raw_size = val_2.get('m2')
        elif val_2: raw_size = val_2

    size_val = 0.0
    if raw_size:
        try:
            clean_str = re.sub(r'[^0-9.]', '', str(raw_size))
            if clean_str: size_val = float(clean_str)
        except: pass

    # 층수
    floor_raw = item.get('floor')
    all_floors_raw = item.get('floor_all')
    fl_curr, fl_all = "", ""
    if isinstance(floor_raw, dict):
        fl_curr = str(floor_raw.get('floor', '')).replace('층', '')
        fl_all = str(floor_raw.get('allFloors', '')).replace('층', '')
    else:
        fl_curr = str(floor_raw).replace('층', '') if floor_raw else ""
        fl_all = str(all_floors_raw).replace('층', '') if all_floors_raw else ""
    if not fl_all and all_floors_raw and not isinstance(all_floors_raw, dict):
         fl_all = str(all_floors_raw).replace('층', '')

    floor_str = f"{fl_curr}/{fl_all}" if fl_all else fl_curr
    
    # -------------------------------------------------------------
    # 🚧 [수정됨] 주차 여부(hasParking) 추출 로직 완벽 수정
    # (API에서 숨겨진 키값과 JSON 전체 문자열 교차 검증)
    # -------------------------------------------------------------
    has_parking = "주차 불가능"
    
    # 1차 시도: API에 존재할 수 있는 다양한 주차 관련 키를 모두 확인
    parking_keys = ['parking', 'parkingYn', 'parking_yn', 'isParking']
    for key in parking_keys:
        val = item.get(key)
        if val is not None:
            # 공백 제거 후 소문자로 변환하여 비교 ("주차 가능" -> "주차가능")
            val_str = str(val).replace(" ", "").lower()
            
            if "불가능" in val_str or val_str in ['0', '0.0', '0대', 'false', '없음', '-', 'none', '']:
                has_parking = "주차 불가능"
            else:
                has_parking = "주차 가능"
            break # 값을 찾았으면 반복문 종료

    # 2차 시도 (교차 검증): 옵션 리스트에 "주차"라는 단어가 명시되어 있다면 가능으로 덮어씀
    if has_parking == "주차 불가능" and options_str and "주차" in options_str:
        has_parking = "주차 가능"

    # 3차 시도 (최후 수단): 만약 API 구조가 특이해서 값을 못 찾았다면,
    # 해당 매물의 전체 JSON 텍스트 데이터를 뒤져서 "주차가능" 키워드가 렌더링될 데이터에 있는지 확인
    if has_parking == "주차 불가능":
        # item 딕셔너리를 통째로 문자열로 변환하고 공백 제거
        item_raw_string = json.dumps(item, ensure_ascii=False).replace(" ", "")
        # 화면에 띄울 데이터(주차가능)가 JSON 어딘가에 존재한다면 주차 가능으로 판단
        if "주차가능" in item_raw_string or '"parking":"가능"' in item_raw_string.replace("'", '"'):
            has_parking = "주차 가능"
    # -------------------------------------------------------------

    loc = item.get('random_location') or item.get('location') or {}

    return {
        'id': item_id,
        'url': item_url,
        'address': jibun,
        'rent_type': sales_type,
        'deposit': deposit,
        'price': price,
        'year_built': year_built,
        'size_m2': size_val,
        'images_count': len(image_links.split('|')) if image_links else 0,
        'images': image_links, 
        'floor': floor_str,
        'hasParking': has_parking,
        'options': options_str,
        'lat': loc.get('lat'),
        'lng': loc.get('lng')
    }

# 5. 메인 실행
# -----------------------------------------------------------
def main():
    # 1. ID 수집 (필터링 없이 일단 다 가져옴)
    target_ids = get_all_ids_raw()
    
    if not target_ids:
        print("❌ 수집된 매물이 없습니다. (네트워크/차단 확인 필요)")
        return

    print(f"\n🚀 [2단계] 셀레니움 상세 수집 및 서울(인천중구제외) 필터링 시작 (대상: {len(target_ids)}개)...")
    driver = get_chrome_driver()
    results = []
    
    try:
        driver.get("https://www.zigbang.com")
        time.sleep(2)
        
        for i, iid in enumerate(target_ids):
            try:
                driver.get(f"https://apis.zigbang.com/v3/items/{iid}")
                
                body = driver.find_element(By.TAG_NAME, "body").text
                if not body.strip().startswith("{"):
                     try: body = driver.find_element(By.TAG_NAME, "pre").text
                     except: pass
                
                data = json.loads(body)
                parsed = parse_detail_data(iid, data)
                
                # parsed가 None이 아니면 수집 대상임
                if parsed:
                    results.append(parsed)
                    # 진행 상황 출력 (50개마다)
                    if len(results) % 50 == 0:
                        print(f"   ✅ {len(results)}번째 매물 저장 중: {parsed['address']} ({i+1}/{len(target_ids)} 탐색)")
                
            except Exception as e:
                pass

        if results:
            df = pd.DataFrame(results)
            # 컬럼 정리 (year_built 추가됨)
            cols = ['id', 'rent_type', 'address', 'deposit', 'price', 'year_built', 'size_m2', 'floor', 'options', 'hasParking', 'images', 'url', 'lat', 'lng']
            final_cols = [c for c in cols if c in df.columns]
            df = df[final_cols]
            
            print(f"\n✨ [최종 완료] 유효 매물 {len(df)}건 수집됨.")
            print(df.head())
            
            filename = "seoul_zigbang_complete12345.csv"
            df.to_csv(filename, index=False, encoding='utf-8-sig')
            print(f"\n💾 '{filename}' 저장 완료!")
        else:
            print("\n❌ 조건에 맞는 매물이 하나도 없습니다.")

    finally:
        driver.quit()

if __name__ == "__main__":
    main()
